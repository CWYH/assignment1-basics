cmake_minimum_required(VERSION 3.20)
project(bpe_benchmark LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_8 ON CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_16 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_32 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2GREP OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PCRE2_SUPPORT_JIT ON CACHE BOOL "" FORCE)
add_subdirectory(../vendor/pcre2-10.46 pcre2)
add_executable(bpe_cpp main.cpp)
target_compile_definitions(bpe_cpp PRIVATE PCRE2_STATIC PCRE2_CODE_UNIT_WIDTH=8)
target_link_libraries(bpe_cpp PRIVATE pcre2-8-static)
if(MINGW)
    target_link_options(bpe_cpp PRIVATE -static-libgcc -static-libstdc++)
endif()
